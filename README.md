    <script>
        // Questions Data Array
        const questionsData = [
            {
                id: 1,
                words: ["prawns", "mussels", "duck", "squid"],
                correct: "duck",
                explanation: "<strong>duck</strong> (pato) é uma ave/carne, enquanto <em>prawns</em>, <em>mussels</em> e <em>squid</em> são frutos do mar."
            },
            {
                id: 2,
                words: ["lamb", "crab", "beef", "pork"],
                correct: "crab",
                explanation: "<strong>crab</strong> (caranguejo) é um fruto do mar, enquanto <em>lamb</em>, <em>beef</em> e <em>pork</em> são carnes de animais terrestres."
            },
            {
                id: 3,
                words: ["cherry", "pear", "peach", "beetroot"],
                correct: "beetroot",
                explanation: "<strong>beetroot</strong> (beterraba) é um vegetal/raiz, enquanto <em>cherry</em>, <em>pear</em> e <em>peach</em> são frutas."
            },
            {
                id: 4,
                words: ["raspberry", "cucumber", "pepper", "cabbage"],
                correct: "raspberry",
                explanation: "<strong>raspberry</strong> (framboesa) é uma fruta/fruta vermelha, enquanto <em>cucumber</em>, <em>pepper</em> e <em>cabbage</em> são vegetais."
            },
            {
                id: 5,
                words: ["fried", "baked", "chicken", "roast"],
                correct: "chicken",
                explanation: "<strong>chicken</strong> (frango) é o alimento/ingrediente, enquanto <em>fried</em>, <em>baked</em> e <em>roast</em> são métodos de preparo/cozimento."
            }
        ];

        let userAnswers = {};
        let isSubmitted = false;

        // Render questions to DOM
        function renderQuestions() {
            const container = document.getElementById('questions-container');
            container.innerHTML = '';

            questionsData.forEach((q, index) => {
                const card = document.createElement('div');
                card.className = "bg-white rounded-2xl p-6 border border-slate-200/80 shadow-sm transition-all duration-200 hover:border-slate-300 relative";
                card.id = `q-card-${q.id}`;

                let optionsHTML = '';
                q.words.forEach(word => {
                    const isChecked = userAnswers[q.id] === word;
                    optionsHTML += `
                        <div class="relative">
                            <input 
                                type="radio" 
                                id="q${q.id}-${word}" 
                                name="question-${q.id}" 
                                value="${word}" 
                                ${isChecked ? 'checked' : ''}
                                ${isSubmitted ? 'disabled' : ''}
                                onchange="selectAnswer(${q.id}, '${word}')"
                                class="custom-radio sr-only"
                            >
                            <label 
                                for="q${q.id}-${word}" 
                                id="label-q${q.id}-${word}"
                                class="flex items-center justify-between p-3.5 rounded-xl border border-slate-200 cursor-pointer hover:bg-slate-50 transition-all font-medium text-slate-700 text-sm sm:text-base capitalize"
                            >
                                <span>${word}</span>
                                <span class="word-dot w-4 h-4 rounded-full border-2 border-slate-300 flex items-center justify-center shrink-0 transition-colors"></span>
                            </label>
                        </div>
                    `;
                });

                card.innerHTML = `
                    <div class="flex items-start justify-between gap-4 mb-4">
                        <div class="flex items-center gap-3">
                            <span class="flex items-center justify-center w-7 h-7 rounded-lg bg-purple-100 text-purple-700 font-bold text-xs">
                                ${index + 1}
                            </span>
                            <h3 class="text-base sm:text-lg font-semibold text-slate-800 leading-snug">
                                Qual palavra é diferente?
                            </h3>
                        </div>
                        <div id="status-icon-${q.id}" class="shrink-0 hidden"></div>
                    </div>
                    <div class="grid grid-cols-2 sm:grid-cols-4 gap-3">
                        ${optionsHTML}
                    </div>
                    <div id="feedback-${q.id}" class="mt-4 text-sm hidden font-medium flex items-start gap-2 p-3.5 rounded-xl"></div>
                `;

                container.appendChild(card);
            });

            lucide.createIcons();
            updateProgress();
        }

        // Handle Radio Selection
        function selectAnswer(questionId, word) {
            if (isSubmitted) return;
            userAnswers[questionId] = word;
            updateProgress();
        }

        // Update progress bar
        function updateProgress() {
            const total = questionsData.length;
            const answeredCount = Object.keys(userAnswers).length;
            const percentage = (answeredCount / total) * 100;

            document.getElementById('progress-text').innerText = `${answeredCount} / ${total} respondidas`;
            document.getElementById('progress-bar').style.width = `${percentage}%`;
        }

        // Submit Quiz
        function submitQuiz(e) {
            e.preventDefault();
            if (isSubmitted) return;

            const total = questionsData.length;
            const answeredCount = Object.keys(userAnswers).length;

            if (answeredCount < total) {
                if (!confirm(`Você respondeu apenas ${answeredCount} de ${total} perguntas. Deseja enviar assim mesmo?`)) {
                    return;
                }
            }

            isSubmitted = true;
            let score = 0;

            questionsData.forEach(q => {
                const card = document.getElementById(`q-card-${q.id}`);
                const statusIcon = document.getElementById(`status-icon-${q.id}`);
                const feedback = document.getElementById(`feedback-${q.id}`);
                const userAnswer = userAnswers[q.id];

                // Disable inputs and style labels
                q.words.forEach(word => {
                    const radio = document.getElementById(`q${q.id}-${word}`);
                    const label = document.getElementById(`label-q${q.id}-${word}`);
                    radio.disabled = true;

                    // Highlight Correct Option always
                    if (word === q.correct) {
                        label.classList.remove('border-slate-200', 'hover:bg-slate-50');
                        label.classList.add('border-emerald-500', 'bg-emerald-50', 'text-emerald-900', 'font-semibold');
                    }

                    // If user selected incorrect option, highlight error
                    if (userAnswer === word && userAnswer !== q.correct) {
                        label.classList.remove('border-slate-200', 'hover:bg-slate-50');
                        label.classList.add('border-red-400', 'bg-red-50', 'text-red-900', 'line-through');
                    }
                });

                statusIcon.classList.remove('hidden');
                feedback.classList.remove('hidden');

                if (userAnswer === q.correct) {
                    score++;
                    card.classList.add('border-emerald-200', 'bg-emerald-50/20');
                    statusIcon.innerHTML = `<span class="flex items-center justify-center w-7 h-7 rounded-full bg-emerald-100 text-emerald-600"><i data-lucide="check" class="w-4 h-4"></i></span>`;
                    feedback.className = "mt-4 text-sm font-medium flex items-start gap-2 p-3.5 rounded-xl bg-emerald-100/60 text-emerald-800";
                    feedback.innerHTML = `<i data-lucide="check-circle-2" class="w-4 h-4 text-emerald-600 shrink-0 mt-0.5"></i> <div><strong>Correto!</strong> ${q.explanation}</div>`;
                } else {
                    card.classList.add('border-red-200', 'bg-red-50/20');
                    statusIcon.innerHTML = `<span class="flex items-center justify-center w-7 h-7 rounded-full bg-red-100 text-red-600"><i data-lucide="x" class="w-4 h-4"></i></span>`;
                    feedback.className = "mt-4 text-sm font-medium flex items-start gap-2 p-3.5 rounded-xl bg-red-100/60 text-red-800";
                    feedback.innerHTML = `<i data-lucide="alert-circle" class="w-4 h-4 text-red-600 shrink-0 mt-0.5"></i> <div><strong>Incorreto.</strong> ${q.explanation}</div>`;
                }
            });

            // Display Score Summary Banner
            const percentage = Math.round((score / total) * 100);
            const banner = document.getElementById('result-banner');
            const scoreDisplay = document.getElementById('score-display');
            const percentageDisplay = document.getElementById('percentage-display');
            const feedbackText = document.getElementById('result-feedback');
            const resultBadge = document.getElementById('result-badge');
            const resultIcon = document.getElementById('result-icon');

            scoreDisplay.innerText = `${score} / ${total}`;
            percentageDisplay.innerText = `${percentage}% de aproveitamento`;

            if (percentage === 100) {
                resultBadge.className = "inline-flex items-center justify-center p-3 rounded-full mb-3 bg-emerald-100 text-emerald-600";
                resultIcon.setAttribute('data-lucide', 'trophy');
                feedbackText.innerText = "Perfeito! Você acertou todas as palavras e domina muito bem esse vocabulário de alimentos e preparos!";
            } else if (percentage >= 60) {
                resultBadge.className = "inline-flex items-center justify-center p-3 rounded-full mb-3 bg-amber-100 text-amber-600";
                resultIcon.setAttribute('data-lucide', 'thumbs-up');
                feedbackText.innerText = "Muito bom! Você demonstrou um bom conhecimento de vocabulário. Confira as explicações nas respostas para revisar.";
            } else {
                resultBadge.className = "inline-flex items-center justify-center p-3 rounded-full mb-3 bg-red-100 text-red-600";
                resultIcon.setAttribute('data-lucide', 'book-open');
                feedbackText.innerText = "Bom tento! Veja as explicações em cada questão para aprender a diferença entre frutas, vegetais, carnes e métodos de cozimento.";
            }

            banner.classList.remove('hidden');
            banner.scrollIntoView({ behavior: 'smooth', block: 'start' });

            // Disable submit button
            const submitBtn = document.getElementById('submit-btn');
            submitBtn.disabled = true;
            submitBtn.classList.add('opacity-50', 'cursor-not-allowed');

            lucide.createIcons();
        }

        // Reset Quiz Logic
        function resetQuiz() {
            userAnswers = {};
            isSubmitted = false;

            document.getElementById('result-banner').classList.add('hidden');
            
            const submitBtn = document.getElementById('submit-btn');
            submitBtn.disabled = false;
            submitBtn.classList.remove('opacity-50', 'cursor-not-allowed');

            renderQuestions();
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // Initialize on page load
        window.onload = function() {
            renderQuestions();
        };
    </script>
